const calculateAge = (birthDateString) => {
  if (!birthDateString) return "-";
  const today = new Date();
  const birthDate = new Date(birthDateString);
  
  let years = today.getFullYear() - birthDate.getFullYear();
  let months = today.getMonth() - birthDate.getMonth();
  let days = today.getDate() - birthDate.getDate();

  if (days < 0) {
    const prevMonth = new Date(today.getFullYear(), today.getMonth(), 0);
    days += prevMonth.getDate();
    months--;
  }
  if (months < 0) {
    months += 12;
    years--;
  }

  const ageParts = [];
  if (years > 0) ageParts.push(`${years} Tahun`);
  if (months > 0) ageParts.push(`${months} Bulan`);
  if (days > 0 || ageParts.length === 0) ageParts.push(`${days} Hari`);

  return ageParts.join(" ");
};

const getAgeInYears = (birthDateString) => {
  if (!birthDateString) return 0;
  const today = new Date();
  const birthDate = new Date(birthDateString);
  let age = today.getFullYear() - birthDate.getFullYear();
  const m = today.getMonth() - birthDate.getMonth();
  if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) age--;
  return age;
};

const getHubunganBadgeColor = (hubungan) => {
  const hub = hubungan?.toLowerCase() || '';
  if (hub.includes('istri')) return 'from-purple-700 to-indigo-900 border-purple-600/30';
  if (hub.includes('anak')) return 'from-blue-600 to-indigo-800 border-blue-600/30';
  if (hub.includes('tua') || hub.includes('ibu') || hub.includes('ayah')) return 'from-amber-600 to-red-800 border-amber-600/30';
  return 'from-teal-600 to-emerald-800 border-teal-600/30'; 
};

function DemografiTab({ daftarWarga }) {
  const [activeSubTab, setActiveTabFilter] = useState('usia'); 
  const [selectedSegment, setSelectedSegment] = useState(null);

  const compileAllIndividuals = () => {
    const list = [];
    daftarWarga.forEach(w => {
      list.push({ id: w.id, name: w.name, address: w.address, status: w.status, gender: w.gender || "Laki-laki", religion: w.religion || "Islam", birthDate: w.birthDate, role: "Kepala Keluarga" });
      if (w.anggotaKeluarga && w.anggotaKeluarga.length > 0) {
        w.anggotaKeluarga.forEach((m, idx) => {
          list.push({ id: `${w.id}-member-${idx}`, name: m.name, address: w.address, status: w.status, gender: m.gender || (m.hubungan?.toLowerCase().includes('istri') ? 'Perempuan' : 'Laki-laki'), religion: m.religion || w.religion || "Islam", birthDate: m.birthDate, role: m.hubungan });
        });
      }
    });
    return list;
  };

  const allIndividuals = compileAllIndividuals();
  const totalJiwa = allIndividuals.length;

  const segments = {
    usia: [{ key: 'balita', label: 'Balita (0-5 Th)', color: 'bg-rose-500' }, { key: 'anak', label: 'Anak (6-12 Th)', color: 'bg-orange-500' }, { key: 'remaja', label: 'Remaja (13-17 Th)', color: 'bg-amber-500' }, { key: 'dewasa', label: 'Dewasa (18-59 Th)', color: 'bg-emerald-500' }, { key: 'lansia', label: 'Lansia (60+ Th)', color: 'bg-indigo-500' }],
    gender: [{ key: 'Laki-laki', label: 'Laki-Laki', color: 'bg-blue-500' }, { key: 'Perempuan', label: 'Perempuan', color: 'bg-pink-500' }],
    status: [{ key: 'Warga Tetap', label: 'Warga Tetap', color: 'bg-teal-500' }, { key: 'Warga Kontrak', label: 'Warga Kontrak', color: 'bg-amber-500' }],
    agama: [{ key: 'Islam', label: 'Islam', color: 'bg-emerald-500' }, { key: 'Kristen', label: 'Kristen', color: 'bg-sky-500' }, { key: 'Katolik', label: 'Katolik', color: 'bg-purple-500' }, { key: 'Hindu', label: 'Hindu', color: 'bg-orange-600' }, { key: 'Budha', label: 'Budha', color: 'bg-yellow-500' }, { key: 'Konghucu', label: 'Konghucu', color: 'bg-red-500' }]
  };

  const getCategorizedList = (category, key) => {
    return allIndividuals.filter(person => {
      if (category === 'usia') { const age = getAgeInYears(person.birthDate); if (key === 'balita') return age <= 5; if (key === 'anak') return age >= 6 && age <= 12; if (key === 'remaja') return age >= 13 && age <= 17; if (key === 'dewasa') return age >= 18 && age <= 59; if (key === 'lansia') return age >= 60; }
      if (category === 'gender') return person.gender === key;
      if (category === 'status') return person.status === key;
      if (category === 'agama') return person.religion === key;
      return false;
    });
  };

  useEffect(() => {
    if (segments[activeSubTab] && segments[activeSubTab].length > 0) {
      setSelectedSegment({ category: activeSubTab, key: segments[activeSubTab][0].key, label: segments[activeSubTab][0].label });
    }
  }, [activeSubTab]);

  const drilldownList = selectedSegment ? getCategorizedList(selectedSegment.category, selectedSegment.key) : [];

  return (
    <div className="p-4 pb-24 border-t border-gray-100 animate-in fade-in duration-200">
      <h2 className="text-xl font-bold text-gray-800 mb-1">Demografi Warga</h2>
      <p className="text-gray-500 text-xs mb-5">Analisis statistik populasi warga RT 03/RW 011 secara real-time.</p>

      <div className="grid grid-cols-4 gap-1.5 mb-5 bg-gray-100 p-1 rounded-xl">
        {[{ id: 'usia', label: 'Usia' }, { id: 'gender', label: 'Gender' }, { id: 'status', label: 'Status' }, { id: 'agama', label: 'Agama' }].map(tab => (
          <button key={tab.id} onClick={() => setActiveTabFilter(tab.id)} className={`py-2 rounded-lg text-[10px] font-black tracking-wide uppercase transition-all ${activeSubTab === tab.id ? 'bg-white text-indigo-600 shadow-sm' : 'text-gray-500 hover:text-gray-700'}`}>{tab.label}</button>
        ))}
      </div>

      <div className="bg-white p-5 rounded-2xl border border-gray-100 shadow-sm mb-6 space-y-4">
        <div className="flex justify-between items-center border-b pb-2 mb-2">
          <h3 className="font-extrabold text-xs text-gray-800 uppercase tracking-wider">Statistik Distribusi</h3>
          <span className="text-[10px] bg-indigo-50 text-indigo-700 px-2 py-0.5 rounded-full font-bold">Total: {totalJiwa} Jiwa</span>
        </div>
        {segments[activeSubTab].map(seg => {
          const count = getCategorizedList(activeSubTab, seg.key).length;
          const pct = totalJiwa > 0 ? Math.round((count / totalJiwa) * 100) : 0;
          return (
            <div key={seg.key} onClick={() => setSelectedSegment({ category: activeSubTab, key: seg.key, label: seg.label })} className={`group cursor-pointer p-2 rounded-xl transition-all ${selectedSegment?.key === seg.key ? 'bg-indigo-50/50 border border-indigo-100' : 'hover:bg-slate-50'}`}>
              <div className="flex justify-between items-center text-xs font-bold text-gray-700 mb-1.5"><span className="flex items-center gap-1.5"><span className={`w-2.5 h-2.5 rounded-full ${seg.color}`}></span>{seg.label}</span><span>{count} Jiwa ({pct}%)</span></div>
              <div className="w-full bg-slate-100 h-2.5 rounded-full overflow-hidden"><div className={`h-full rounded-full ${seg.color} transition-all duration-500`} style={{ width: pct + '%' }}></div></div>
            </div>
          );
        })}
      </div>

      {selectedSegment && (
        <div className="animate-in fade-in slide-in-from-bottom-2 duration-150">
          <div className="flex justify-between items-center mb-3">
            <h3 className="font-bold text-sm text-gray-800 flex items-center gap-1"><Users size={16} className="text-indigo-600" /> Detail: {selectedSegment.label}</h3>
            <span className="text-[10px] font-black bg-indigo-100 text-indigo-800 px-2.5 py-0.5 rounded-full">{drilldownList.length} Jiwa</span>
          </div>
          <div className="space-y-2">
            {drilldownList.map(person => (
              <div key={person.id} className="bg-white p-3 rounded-xl border border-gray-100 shadow-sm flex justify-between items-center gap-3">
                <div className="min-w-0"><h4 className="font-extrabold text-xs text-gray-800 truncate">{person.name}</h4><p className="text-[10px] text-gray-400 mt-0.5 truncate">{person.address} | <span className="font-semibold text-gray-600">{person.role}</span></p></div>
                <div className="text-right shrink-0 flex flex-col items-end gap-1"><span className="text-[9px] font-black text-indigo-700 bg-indigo-50 px-2 py-0.5 rounded-full uppercase">{calculateAge(person.birthDate)}</span></div>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

      });

  const intToRoman = (value) => {
    if (value <= 0) return '0';
    const romanNumerals = [
      [1000, 'M'], [900, 'CM'], [500, 'D'], [400, 'CD'],
      [100, 'C'], [90, 'XC'], [50, 'L'], [40, 'XL'],
      [10, 'X'], [9, 'IX'], [5, 'V'], [4, 'IV'], [1, 'I'],
    ];
    let result = '';
    let num = value;
    for (const [intVal, roman] of romanNumerals) {
      while (num >= intVal) {
        result += roman;
        num -= intVal;
      }
    }
    return result;
  };

      }
    }, "Tambahkan");
  };

    setEditingWargaId(w.id);
    setNewWarga({
      name: w.name || '',
      nik: w.nik || '',
      address: w.address || '',
      status: w.status || 'Warga Tetap',

      birthDate: w.birthDate || '',
      gender: w.gender || 'Laki-laki',
      religion: w.religion || 'Islam'

      isEdit ? "Perubahan data warga akan disimpan ke sistem." : "Data Anda akan diproses dan disimpan secara permanen ke Firebase Cloud.",

    localStorage.setItem('rtKopType', rtKopType);
    localStorage.setItem('rtKopLogo', rtKopLogo);
    localStorage.setItem('rtKopBanner', rtKopBanner);
    localStorage.setItem('rtSignature', rtSignature);
    localStorage.setItem('rtStamp', rtStamp);
    localStorage.setItem('rwLeaderName', rwLeaderName);
  }, [rtKopType, rtKopLogo, rtKopBanner, rtSignature, rtStamp, rwLeaderName]);

  if (!isLoggedIn) return <LoginScreen onLogin={handleLogin} adminPassword={adminPassword} daftarWarga={daftarWarga} />;

  const allCitizensForSync = [];
  daftarWarga.forEach(w => {
    allCitizensForSync.push({ ...w, refId: w.id.toString() });
    if (w.anggotaKeluarga) {
      w.anggotaKeluarga.forEach((m, idx) => {
        allCitizensForSync.push({ ...m, refId: `${w.id}_${idx}`, address: w.address });
      });
    }
  });

  const enrichedPengurus = pengurusRT.map(p => {
    const linkedWarga = allCitizensForSync.find(c => c.refId === p.wargaId || c.name === p.name);
    return {
      ...p,
      name: linkedWarga ? linkedWarga.name : p.name,
      photo: linkedWarga ? (linkedWarga.photo || p.photo) : p.photo
    };
  });

  return (
    <div className="min-h-screen bg-gray-100 flex items-center justify-center p-0 md:p-4 font-sans antialiased relative">
      {/* PITA PERINGATAN SERVER */}
      {serverError && (
        <div className="absolute top-0 left-0 w-full bg-red-600 text-white p-3 text-center text-xs font-bold z-[9999] shadow-md flex justify-between items-center px-4 animate-in slide-in-from-top-4">
          <div className="flex items-center gap-2">
            <AlertTriangle size={16} /> <span>{serverError}</span>
          </div>
          <button onClick={() => setServerError(null)} className="p-1 hover:bg-red-700 rounded-full transition-colors"><X size={16} /></button>
        </div>
      )}

      <div className="w-full max-w-[400px] h-[100dvh] md:h-[850px] bg-white md:rounded-[2.5rem] shadow-2xl relative overflow-hidden flex flex-col md:border-[10px] md:border-gray-950">
        <header className="bg-white px-3 py-2 border-b flex justify-between items-center min-h-[64px] shrink-0">
          <img src="/image_385930.png" alt="Kop" className="h-12 object-contain" onError={({ currentTarget }) => { currentTarget.style.display = 'none'; }} />
          <button onClick={handleLogout} className="text-red-500 bg-red-50 px-3 py-1.5 rounded-lg hover:bg-red-100 transition-colors flex items-center gap-1.5 text-[10px] font-black uppercase tracking-wide">
            <LogOut size={12} /> Keluar
          </button>
        </header>

        <main className="flex-1 overflow-y-auto bg-gray-50/50 hide-scrollbar relative">
          {activeTab === 'home' && <HomeTab currentUserProfile={currentUserProfile} setCurrentUserProfile={setCurrentUserProfile} pengurusRT={enrichedPengurus} announcements={announcements} daftarWarga={daftarWarga} complaints={complaints} isAdmin={isAdmin} onReplyComplaint={handleReplyComplaint} onDeleteComplaint={handleDeleteComplaint} setActiveTab={setActiveTab} showModal={showModal} setSelectedOfficer={setSelectedOfficer} setSelectedBirthdayWarga={setSelectedBirthdayWarga} setIsComplaintModalOpen={setIsComplaintModalOpen} setPreviewImage={setPreviewImage} db={db} appId={appId} />}
          {activeTab === 'surat' && <SuratTab currentUserProfile={currentUserProfile} suratRequests={suratRequests} setSelectedPrintSurat={setSelectedPrintSurat} showModal={showModal} rtKopType={rtKopType} rtKopLogo={rtKopLogo} rtKopBanner={rtKopBanner} rwLeaderName={rwLeaderName} db={db} appId={appId} showConfirm={showConfirm} handleDeleteSurat={handleDeleteSurat} />}
          {activeTab === 'warga' && <WargaTab daftarWarga={daftarWarga} setSelectedWargaDetail={setSelectedWargaDetail} />}
          {activeTab === 'demografi' && <DemografiTab daftarWarga={daftarWarga} />}
          {activeTab === 'admin' && isAdmin && <AdminTab suratRequests={suratRequests} pengurusRT={enrichedPengurus} daftarWarga={daftarWarga} complaints={complaints} announcements={announcements} setIsAdmin={() => setIsLoggedIn(false)} setActiveTab={setActiveTab} showModal={showModal} setSelectedPrintSurat={setSelectedPrintSurat} rtSignature={rtSignature} setRtSignature={setRtSignature} rtStamp={rtStamp} setRtStamp={setRtStamp} rtKopType={rtKopType} setRtKopType={setRtKopType} rtKopLogo={rtKopLogo} setRtKopLogo={setRtKopLogo} rtKopBanner={rtKopBanner} setRtKopBanner={setRtKopBanner} rwLeaderName={rwLeaderName} setRwLeaderName={setRwLeaderName} db={db} appId={appId} showConfirm={showConfirm} adminPassword={adminPassword} />}
        </main>

        <nav className="bg-white border-t border-gray-100 pb-2 md:pb-0 z-20 shrink-0 shadow-inner">
          <div className="flex justify-around items-center h-16 px-1">
            <button onClick={() => { setActiveTab('home'); }} className={`flex flex-col items-center justify-center w-1/5 h-full space-y-1 ${activeTab === 'home' ? 'text-blue-600' : 'text-gray-400'}`}><Home size={18} /> <span className="text-[8px] font-bold">Beranda</span></button>
            <button onClick={() => { setActiveTab('surat'); }} className={`flex flex-col items-center justify-center w-1/5 h-full space-y-1 ${activeTab === 'surat' ? 'text-blue-600' : 'text-gray-400'}`}><FileText size={18} /> <span className="text-[8px] font-bold">Surat</span></button>
            <button onClick={() => { setActiveTab('warga'); }} className={`flex flex-col items-center justify-center w-1/5 h-full space-y-1 ${activeTab === 'warga' ? 'text-blue-600' : 'text-gray-400'}`}><Users size={18} /> <span className="text-[8px] font-bold">Warga</span></button>
            <button onClick={() => { setActiveTab('demografi'); }} className={`flex flex-col items-center justify-center w-1/5 h-full space-y-1 ${activeTab === 'demografi' ? 'text-blue-600' : 'text-gray-400'}`}><BarChart3 size={18} /> <span className="text-[8px] font-bold">Demografi</span></button>
            {isAdmin && <button onClick={() => { setActiveTab('admin'); }} className={`flex flex-col items-center justify-center w-1/5 h-full space-y-1 ${activeTab === 'admin' ? 'text-blue-600' : 'text-gray-400'}`}><Settings size={18} /> <span className="text-[8px] font-bold">Admin</span></button>}
          </div>
          <div className="w-full text-center py-1.5 bg-gray-50 border-t border-gray-100 hidden md:block"><p className="text-[9px] text-gray-400 font-medium">© 2026 R.S Program Development</p></div>
        </nav>

        <CustomModal isOpen={modalConfig.isOpen} title={modalConfig.title} message={modalConfig.message} type={modalConfig.type} onClose={closeModal} />
        <ConfirmPopup isOpen={confirmConfig.isOpen} title={confirmConfig.title} message={confirmConfig.message} onConfirm={confirmConfig.action} onCancel={closeConfirm} confirmText={confirmConfig.confirmText} />
        <ImagePreviewModal imageUrl={previewImage} onClose={() => setPreviewImage(null)} />
        <PrintableLetterModal selectedPrintSurat={selectedPrintSurat} setSelectedPrintSurat={setSelectedPrintSurat} currentUserProfile={currentUserProfile} rwLeaderName={rwLeaderName} rtSignature={rtSignature} rtStamp={rtStamp} rtKopType={rtKopType} rtKopLogo={rtKopLogo} rtKopBanner={rtKopBanner} />
        <OfficerDetailModal officer={selectedOfficer} onClose={() => setSelectedOfficer(null)} onImageClick={setPreviewImage} />
        <WargaDetailModal warga={selectedWargaDetail} onClose={() => setSelectedWargaDetail(null)} isAdmin={isAdmin} onImageClick={setPreviewImage} />
        <BirthdayWishModal isOpen={selectedBirthdayWarga !== null} warga={selectedBirthdayWarga} onClose={() => setSelectedBirthdayWarga(null)} />
        <ComplaintModal isOpen={isComplaintModalOpen} onClose={() => setIsComplaintModalOpen(false)} onSubmit={() => { showModal("Pengaduan Terkirim!", "Aduan Anda dikirim dan akan ditindaklanjuti.", "success"); }} currentUserProfile={currentUserProfile} db={db} appId={appId} showConfirm={showConfirm} showModalLocal={showModal} />
      </div>
    </div>
  );
}
